
module tb;

  reg clk;
  reg rst;
  wire [2:0] count;

  counter uut (
    .clk(clk),
    .rst(rst),
    .count(count)
  );

  initial clk = 0;
  always #5 clk = ~clk;

  initial begin
    $monitor("Time=%0t rst=%b count=%d",
             $time, rst, count);

    rst = 1'b1;
    #2 rst = 1'b0;
    #10 rst = 1'b1;

    repeat (10)
      @(posedge clk);

    #2 $finish;
  end

endmodule
